WITH active_macro AS (
  SELECT mc.id_macrociclo
  FROM adt.macrociclos mc
  WHERE mc.id_alumno = {{Number(sel_routine_student.selectedOptionValue) || 0}}
    AND mc.esta_activo IS TRUE
  ORDER BY mc.fecha_inicio DESC NULLS LAST, mc.id_macrociclo DESC
  LIMIT 1
),
active_meso AS (
  SELECT meso.id_mesociclo, meso.fase_tipo
  FROM adt.mesociclos meso
  JOIN active_macro mc ON mc.id_macrociclo = meso.id_macrociclo
  WHERE meso.esta_activo IS TRUE
  ORDER BY meso.fecha_inicio DESC NULLS LAST, meso.id_mesociclo DESC
  LIMIT 1
)
SELECT
  a.rut,
  mc.id_macrociclo,
  meso.id_mesociclo,
  b.id_bloque,
  b.semana_meso AS semana,
  b.dia_numero AS dia,
  b.dia_tipo,
  b.orden,
  COALESCE(e.nombre, 'Ejercicio #' || b.id_ejercicio::text) AS ejercicio,
  b.series AS series_plan,
  CASE
    WHEN b.reps_min IS NULL AND b.reps_max IS NULL THEN NULL
    WHEN b.reps_min = b.reps_max THEN b.reps_min::text
    ELSE COALESCE(b.reps_min::text, 'N/D') || ' - ' || COALESCE(b.reps_max::text, 'N/D')
  END AS repeticiones_plan,
  COALESCE(
    CASE
      WHEN b.rir_objetivo_min IS NOT NULL AND b.rir_objetivo_max IS NOT NULL
        THEN b.rir_objetivo_min::text || ' - ' || b.rir_objetivo_max::text
    END,
    b.rir_objetivo::text
  ) AS rir_plan,
  b.peso_sugerido_inicial AS peso_sugerido,
  ultimo_log.peso_utilizado AS peso_realizado,
  ultimo_log.series_realizadas,
  ultimo_log.reps_promedio,
  ultimo_log.rir_reportado,
  ultimo_log.rpe_reportado,
  ultimo_log.fecha_realizada AS ultima_ejecucion,
  e.patron_movimiento,
  e.maquina,
  b.objetivo_muscular,
  e.es_primario,
  e.es_secundario,
  e.es_terciario,
  e.grupo_muscular_primario,
  e.grupo_muscular_secundario,
  e.grupo_muscular_terciario
FROM adt.alumnos a
JOIN active_macro mc ON TRUE
JOIN active_meso meso ON TRUE
JOIN adt.bloques_ejercicios b ON b.id_mesociclo = meso.id_mesociclo
LEFT JOIN adt.ejercicios e ON e.id_ejercicio = b.id_ejercicio
LEFT JOIN LATERAL (
  SELECT
    le.peso_utilizado,
    le.series_realizadas,
    le.reps_promedio,
    le.rir_reportado,
    le.rpe_reportado,
    le.fecha_realizada
  FROM adt.log_entrenamiento le
  WHERE le.id_bloque = b.id_bloque
  ORDER BY le.fecha_realizada DESC, le.id_log DESC
  LIMIT 1
) ultimo_log ON TRUE
WHERE a.id_alumno = {{Number(sel_routine_student.selectedOptionValue) || 0}}
ORDER BY b.semana_meso, b.dia_numero, b.orden;
