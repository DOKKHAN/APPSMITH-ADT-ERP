INSERT INTO adt.log_entrenamiento (
  id_bloque,
  semana_del_meso,
  fecha_realizada,
  iso_year,
  iso_semana,
  series_realizadas,
  peso_utilizado,
  reps_promedio,
  rir_reportado,
  rpe_reportado
)
SELECT
  b.id_bloque,
  b.semana_meso,
  CURRENT_TIMESTAMP,
  EXTRACT(ISOYEAR FROM CURRENT_DATE)::integer,
  EXTRACT(WEEK FROM CURRENT_DATE)::integer,
  {{tbl_routine_exercises.updatedRow.series_realizadas === '' || tbl_routine_exercises.updatedRow.series_realizadas == null ? null : Number(tbl_routine_exercises.updatedRow.series_realizadas)}},
  {{tbl_routine_exercises.updatedRow.peso_realizado === '' || tbl_routine_exercises.updatedRow.peso_realizado == null ? null : Number(tbl_routine_exercises.updatedRow.peso_realizado)}},
  {{tbl_routine_exercises.updatedRow.reps_promedio === '' || tbl_routine_exercises.updatedRow.reps_promedio == null ? null : Number(tbl_routine_exercises.updatedRow.reps_promedio)}},
  {{tbl_routine_exercises.updatedRow.rir_reportado === '' || tbl_routine_exercises.updatedRow.rir_reportado == null ? null : Number(tbl_routine_exercises.updatedRow.rir_reportado)}},
  {{tbl_routine_exercises.updatedRow.rpe_reportado === '' || tbl_routine_exercises.updatedRow.rpe_reportado == null ? null : Number(tbl_routine_exercises.updatedRow.rpe_reportado)}}
FROM adt.bloques_ejercicios b
JOIN adt.mesociclos meso ON meso.id_mesociclo = b.id_mesociclo
JOIN adt.macrociclos mc ON mc.id_macrociclo = meso.id_macrociclo
WHERE b.id_bloque = {{Number(tbl_routine_exercises.updatedRow.id_bloque) || 0}}
  AND mc.id_alumno = {{Number(sel_routine_student.selectedOptionValue) || 0}}
  AND mc.esta_activo IS TRUE
  AND meso.esta_activo IS TRUE
  AND {{tbl_routine_exercises.updatedRow.peso_realizado === '' || tbl_routine_exercises.updatedRow.peso_realizado == null ? null : Number(tbl_routine_exercises.updatedRow.peso_realizado)}} IS NOT NULL
  AND {{Number(tbl_routine_exercises.updatedRow.peso_realizado)}} BETWEEN 0 AND 1000
  AND ({{tbl_routine_exercises.updatedRow.series_realizadas === '' || tbl_routine_exercises.updatedRow.series_realizadas == null ? null : Number(tbl_routine_exercises.updatedRow.series_realizadas)}} IS NULL OR {{Number(tbl_routine_exercises.updatedRow.series_realizadas)}} BETWEEN 1 AND 20)
  AND ({{tbl_routine_exercises.updatedRow.reps_promedio === '' || tbl_routine_exercises.updatedRow.reps_promedio == null ? null : Number(tbl_routine_exercises.updatedRow.reps_promedio)}} IS NULL OR {{Number(tbl_routine_exercises.updatedRow.reps_promedio)}} BETWEEN 0 AND 1000)
  AND ({{tbl_routine_exercises.updatedRow.rir_reportado === '' || tbl_routine_exercises.updatedRow.rir_reportado == null ? null : Number(tbl_routine_exercises.updatedRow.rir_reportado)}} IS NULL OR {{Number(tbl_routine_exercises.updatedRow.rir_reportado)}} BETWEEN 0 AND 10)
  AND ({{tbl_routine_exercises.updatedRow.rpe_reportado === '' || tbl_routine_exercises.updatedRow.rpe_reportado == null ? null : Number(tbl_routine_exercises.updatedRow.rpe_reportado)}} IS NULL OR {{Number(tbl_routine_exercises.updatedRow.rpe_reportado)}} BETWEEN 0 AND 10)
RETURNING id_log, id_bloque, fecha_realizada, peso_utilizado;
