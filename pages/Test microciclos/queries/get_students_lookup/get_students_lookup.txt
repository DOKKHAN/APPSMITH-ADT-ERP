SELECT
  id_alumno AS value,
  COALESCE(NULLIF(rut, ''), 'Sin RUT') || ' - ' || nombre_completo AS label,
  id_alumno,
  nombre_completo,
  COALESCE(NULLIF(rut, ''), 'Sin RUT') AS rut,
  frecuencia_semanal,
  objetivo,
  prioridad_recomposicion,
  enfoque_principal,
  enfoque_especifico,
  debilidad
FROM adt.alumnos
WHERE estado_activo IS TRUE
  AND evaluacion_inicial IS TRUE
ORDER BY rut ASC, nombre_completo ASC;
