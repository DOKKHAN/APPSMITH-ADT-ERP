SELECT
  b.id_bloque,
  b.id_mesociclo,
  b.semana_meso,
  b.dia_numero,
  b.orden,
  e.nombre AS ejercicio,
  NULL::numeric AS reps_promedio,
  NULL::numeric AS peso_utilizado,
  CONCAT(b.reps_min, ' - ', b.reps_max) AS repeticiones,
  b.series,
  b.rir_objetivo_min,
  b.rir_objetivo_max,
  CONCAT(b.rir_objetivo_min, ' - ', b.rir_objetivo_max) AS rir_objetivo,
  b.estado_microciclo,
  CASE
    WHEN b.comentario_semanal ILIKE '%Actualizador nocturno:%'
      THEN LEFT(trim(
        regexp_replace(
          substring(b.comentario_semanal from 'Actualizador nocturno:.*'),
          '[[:space:]]*\|[[:space:]]*(Base mesociclo estable|Matriz:|Orden matriz:|Slot:|Patron:|Musculo:|RIR objetivo).*',
          '',
          'gi'
        )
      ), 180)
    WHEN b.comentario_semanal ILIKE '%promedio reps%'
      OR b.comentario_semanal ILIKE '%estado%'
      OR b.comentario_semanal ILIKE '%Aerobico%'
      OR b.comentario_semanal ILIKE '%Bandas%'
      OR b.comentario_semanal ILIKE '%isometrica%'
      OR b.comentario_semanal ILIKE '%semana % -> %'
      THEN LEFT(trim(
        regexp_replace(
          b.comentario_semanal,
          '[[:space:]]*\|[[:space:]]*(Base mesociclo estable|Matriz:|Orden matriz:|Slot:|Patron:|Musculo:|RIR objetivo).*',
          '',
          'gi'
        )
      ), 180)
    WHEN b.estado_microciclo IS NOT NULL
      THEN b.estado_microciclo
    ELSE NULL
  END AS comentario_microciclo,
  b.actualizado_en
FROM adt.bloques_ejercicios b
JOIN adt.ejercicios e ON e.id_ejercicio = b.id_ejercicio
WHERE b.id_mesociclo = {{Number(sel_test_microcycle.selectedOptionValue) || 0}}
  AND b.semana_meso = {{Number(sel_test_week.selectedOptionValue || appsmith.store.test_microcycle_selected_week || 1)}}
ORDER BY b.dia_numero, b.orden;
