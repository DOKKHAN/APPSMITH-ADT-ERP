WITH input_rows AS (
  SELECT *
  FROM jsonb_to_recordset({{JSON.stringify((tbl_test_microcycle_exercises.processedTableData || tbl_test_microcycle_exercises.updatedRows || []).filter((row) => row.reps_promedio !== '' && row.reps_promedio != null && !Number.isNaN(Number(row.reps_promedio))))}}::jsonb) AS x(
    id_bloque int,
    semana_meso int,
    series int,
    reps_promedio numeric,
    rir_objetivo_max int
  )
  WHERE x.reps_promedio IS NOT NULL
), fecha_simulada AS (
  SELECT
    m.fecha_inicio::date
      + (({{Number(sel_test_week.selectedOptionValue || appsmith.store.test_microcycle_selected_week || 1)}}::int - 1) * INTERVAL '7 days')
      + INTERVAL '12 hours' AS fecha_realizada
  FROM adt.mesociclos m
  WHERE m.id_mesociclo = {{Number(sel_test_microcycle.selectedOptionValue) || 0}}
), limpiar AS (
  DELETE FROM adt.log_entrenamiento l
  USING input_rows i
  WHERE l.id_bloque = i.id_bloque
    AND l.semana_del_meso = i.semana_meso
    AND l.comentario_alumno = 'Prueba Appsmith microciclo'
)
INSERT INTO adt.log_entrenamiento (
  id_bloque,
  semana_del_meso,
  fecha_realizada,
  iso_year,
  iso_semana,
  series_realizadas,
  peso_utilizado,
  reps_logradas,
  reps_promedio,
  rir_reportado,
  rpe_reportado,
  comentario_alumno
)
SELECT
  i.id_bloque,
  i.semana_meso,
  f.fecha_realizada,
  EXTRACT(ISOYEAR FROM f.fecha_realizada)::integer,
  EXTRACT(WEEK FROM f.fecha_realizada)::integer,
  i.series,
  0,
  i.reps_promedio,
  i.reps_promedio,
  i.rir_objetivo_max,
  7,
  'Prueba Appsmith microciclo'
FROM input_rows i
CROSS JOIN fecha_simulada f
RETURNING id_log, id_bloque, semana_del_meso, reps_promedio;
